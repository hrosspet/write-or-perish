# Staging environment: staging.loore.org
# Proxies to Docker Compose stack (project: wop-staging)
#   Backend:  127.0.0.1:5011
#   Frontend: 127.0.0.1:8081

# Open live-update streams per client address, as in configs/nginx.txt;
# a zone of its own (both files are loaded into one nginx).
limit_conn_zone $binary_remote_addr zone=loore_staging_sse_per_address:10m;

# The access log: one line per request with the address, time, method, path,
# protocol, status, size and client. The path is written without the part
# after the question mark, and the referring page's address is left out.
# Names of its own, as for the zone above.
map $request_uri $loore_staging_request_path {
    "~^(?<loore_staging_path_only>[^?]*)" $loore_staging_path_only;
    default "";
}
log_format loore_staging_no_query '$remote_addr - $remote_user [$time_local] '
    '"$request_method $loore_staging_request_path $server_protocol" '
    '$status $body_bytes_sent "$http_user_agent"';

server {
   server_name staging.loore.org;
   access_log /var/log/nginx/access.log loore_staging_no_query;

   # Don't leak the nginx version/OS in the Server response header (#142).
   server_tokens off;

   # Request bodies: 2 MB unless a location below allows more; the same
   # sizes as production (configs/nginx.txt).
   client_max_body_size 2m;

   # Proxy media requests through Flask, which checks who may fetch each
   # file, decrypts .enc files and sets Cache-Control (private for files
   # only signed-in users may fetch). Do not add a Cache-Control header here.
   location /media/ {
     proxy_pass http://127.0.0.1:5011;
     proxy_set_header Host $host;
     proxy_set_header X-Forwarded-Host $host;
     proxy_set_header X-Real-IP $remote_addr;
     proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
     proxy_set_header X-Forwarded-Proto $scheme;
     proxy_buffering off;
   }

   # Proxy backend authentication endpoints
   location /auth/ {
       proxy_pass http://127.0.0.1:5011;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
   }

   # SSE (Server-Sent Events) endpoints - require special handling
   location /api/sse/ {
       proxy_pass http://127.0.0.1:5011;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;

       # Critical SSE settings - disable buffering for real-time streaming
       proxy_buffering off;
       proxy_cache off;

       # SSE connections can last up to 2 hours (for long recordings)
       proxy_connect_timeout 60s;
       proxy_send_timeout 7200s;
       proxy_read_timeout 7200s;

       # Ensure chunked encoding works properly
       chunked_transfer_encoding on;

       # Disable keepalive timeouts for long-lived connections
       keepalive_timeout 0;

       limit_conn loore_staging_sse_per_address 400;
       limit_conn_status 429;
   }

   # The data export, sent while the app builds it (#591); see
   # configs/nginx.txt.
   location = /api/export/threads {
       proxy_pass http://127.0.0.1:5011;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
       proxy_http_version 1.1;
       proxy_buffering off;
       proxy_connect_timeout 60s;
       proxy_send_timeout 60s;
       proxy_read_timeout 60s;
   }

   location /api/ {
       proxy_pass http://127.0.0.1:5011;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;

       proxy_connect_timeout 60s;
       proxy_send_timeout 60s;
       proxy_read_timeout 60s;

       # An entry's whole text, its draft, the documents edited in place
       # (16 MB), and the recording and upload chunks (8 MB).
       client_max_body_size 16m;
   }

   # The markdown import's confirm posts the analyzed text back (up to
   # 64 MB of notes with its escaping, 96 MB).
   location = /api/import/confirm {
       proxy_pass http://127.0.0.1:5011;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
       proxy_connect_timeout 60s;
       proxy_send_timeout 60s;
       proxy_read_timeout 60s;
       client_max_body_size 96m;
   }

   # File uploads: an audio file for a new entry, import archives and
   # exports, a bookmarks file (200 MB files, and the form around them).
   location ~ "^/api/(?:nodes/?|import/(?:twitter/|claude/|chatgpt/)?analyze|external/bookmarks/import)$" {
       proxy_pass http://127.0.0.1:5011;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
       proxy_connect_timeout 60s;
       proxy_send_timeout 60s;
       proxy_read_timeout 60s;
       client_max_body_size 201m;
   }

   # Server-rendered public pages (SEO/no-JS pass) — same as production
   # (configs/nginx.txt) but proxying to the staging backend container.
   # On backend failure, fall through to the frontend container's shell.
   location ~ "^/(?:@|node/[0-9]+/?$)|^/(?:sitemap\.xml|landing|why-loore|vision|how-to|login|alpha-thank-you)?$" {
       proxy_pass http://127.0.0.1:5011;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
       proxy_intercept_errors on;
       error_page 502 503 504 =200 /index.html;
   }

   # Everything else → staging frontend nginx container
   location / {
       proxy_pass http://127.0.0.1:8081;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
   }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/staging.loore.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/staging.loore.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}

# HTTP -> HTTPS redirect for staging.loore.org
server {
    if ($host = staging.loore.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    listen 80;
    server_name staging.loore.org;
   access_log /var/log/nginx/access.log loore_staging_no_query;
    return 404; # managed by Certbot
}
