# Primary domain: loore.org

# Open live-update streams (/api/sse/) per client address, under the app's
# own limit per account (backend/utils/open_streams.py). The zone's name
# must differ from staging's: both files are loaded into one nginx.
limit_conn_zone $binary_remote_addr zone=loore_sse_per_address:10m;

server {
   server_name loore.org www.loore.org;
   root /home/hrosspet/write-or-perish/frontend/build;
   index index.html index.htm;

   # Don't leak the nginx version/OS in the Server response header (#142).
   server_tokens off;

   # Request bodies: 2 MB unless a location below allows more. The app
   # holds each route to the same sizes and refuses a larger body itself
   # (backend/utils/request_size.py); these keep it from reaching the app.
   client_max_body_size 2m;

   # Proxy media requests through Flask, which checks who may fetch each
   # file, decrypts .enc files and sets Cache-Control (private for files
   # only signed-in users may fetch). Do not add a Cache-Control header here.
   location /media/ {
     proxy_pass http://127.0.0.1:8000;
     proxy_set_header Host $host;
     proxy_set_header X-Forwarded-Host $host;
     proxy_set_header X-Real-IP $remote_addr;
     proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
     proxy_set_header X-Forwarded-Proto $scheme;
     proxy_buffering off;
   }

   location / {
       try_files $uri $uri/ /index.html;
   }

   # Server-rendered public pages (SEO/no-JS pass): Flask injects the
   # article content and per-route metadata into the SPA shell for
   # articles (/@user/slug), author pages (/@user), /node/<id>, the
   # sitemap, and the marketing pages. If the backend is down, these
   # degrade to the static SPA shell instead of a 502.
   location ~ "^/(?:@|node/[0-9]+/?$)|^/(?:sitemap\.xml|landing|why-loore|vision|how-to|login|alpha-thank-you)?$" {
       proxy_pass http://127.0.0.1:8000;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
       proxy_intercept_errors on;
       error_page 502 503 504 =200 /index.html;
   }

   # Proxy backend authentication endpoints to gunicorn.
   location /auth/ {
       proxy_pass http://127.0.0.1:8000;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
   }

   # SSE (Server-Sent Events) endpoints - require special handling
   location /api/sse/ {
       proxy_pass http://127.0.0.1:8000;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;

       # Critical SSE settings - disable buffering for real-time streaming
       proxy_buffering off;
       proxy_cache off;

       # SSE connections can last up to 2 hours (for long recordings)
       proxy_connect_timeout 60s;
       proxy_send_timeout 7200s;
       proxy_read_timeout 7200s;

       # Ensure chunked encoding works properly
       chunked_transfer_encoding on;

       # Disable keepalive timeouts for long-lived connections
       keepalive_timeout 0;

       # Ten busy accounts behind one address (an office, a campus, a
       # carrier's shared address), at the app's 40 streams each.
       limit_conn loore_sse_per_address 400;
       limit_conn_status 429;
   }

   # The data export is sent while the app builds it, for minutes on a
   # large account (#591). HTTP/1.1 to the app server, so a build that
   # fails part-way cuts the download (the browser marks it failed)
   # instead of ending the file early as if it were complete: under
   # HTTP/1.0 the end of a response is the closed connection, the same
   # for a failure. No buffering, so each piece goes out as it is built.
   # The read timeout is the limit between two pieces; the app keeps
   # them seconds apart.
   location = /api/export/threads {
       proxy_pass http://127.0.0.1:8000;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
       proxy_http_version 1.1;
       proxy_buffering off;
       proxy_connect_timeout 60s;
       proxy_send_timeout 60s;
       proxy_read_timeout 60s;
   }

   location /api/ {
       proxy_pass http://127.0.0.1:8000;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;

       # Timeout settings for API requests (async operations now handled by Celery)
       proxy_connect_timeout 60s;
       proxy_send_timeout 60s;
       proxy_read_timeout 60s;

       # An entry's whole text, its draft, the documents edited in place
       # (16 MB), and the recording and upload chunks (8 MB).
       client_max_body_size 16m;
   }

   # The markdown import's confirm posts the analyzed text back (up to
   # 64 MB of notes with its escaping, 96 MB).
   location = /api/import/confirm {
       proxy_pass http://127.0.0.1:8000;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
       proxy_connect_timeout 60s;
       proxy_send_timeout 60s;
       proxy_read_timeout 60s;
       client_max_body_size 96m;
   }

   # File uploads: an audio file for a new entry, import archives and
   # exports, a bookmarks file (200 MB files, and the form around them).
   location ~ "^/api/(?:nodes/?|import/(?:twitter/|claude/|chatgpt/)?analyze|external/bookmarks/import)$" {
       proxy_pass http://127.0.0.1:8000;
       proxy_set_header Host $host;
       proxy_set_header X-Forwarded-Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
       proxy_connect_timeout 60s;
       proxy_send_timeout 60s;
       proxy_read_timeout 60s;
       client_max_body_size 201m;
   }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/loore.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/loore.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot


}

# HTTP -> HTTPS redirect for loore.org
server {
    if ($host = www.loore.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    if ($host = loore.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


   listen 80;
   server_name loore.org www.loore.org;
    return 404; # managed by Certbot


}
